法线纹理通常是放在切线空间下面的。切线空间展开的法线贴图和主帖图等其他都是展uv的过程中得到的。
通常情况下会把光照方向和视角方向变化到切线空间的坐标系下，或者选择将法线方向转到世界空间下，参与世界空间的光照和视角计算。第一种的效率更高，而第二种的通用性更好。

##### 第一种:切线空间的光照
	struct av2

            {

                float4 vertex : POSITION;

                float2 uv : TEXCOORD0;

                float3 normal : NORMAL;

                float4 tangent : TANGENT;

            };

  

            struct v2f

            {

                float4 pos : SV_POSITION;

                float4 uv : TEXCOORD0;

                float3 LightDir : TEXCOORD1;

                float3 ViewDir : TEXCOORD2;

            };

  

            float4 _MainTex_ST;

            sampler2D _MainTex;

            sampler2D _BumpMap;

            float4 _BumpMap_ST;

            float _BumpScale;

            float _Gloss;

            float4 _Specular;

  

            v2f vert (av2 v) {

                v2f o;

                o.pos = UnityObjectToClipPos(v.vertex);

                o.uv.xy = v.uv.xy * _MainTex_ST.xy + _MainTex_ST.zw;

                o.uv.zw = v.uv.xy * _BumpMap_ST.xy + _BumpMap_ST.zw;

                float3 tangent = normalize(v.tangent.xyz);

                float3 binormal = cross(normalize(v.normal), tangent) * v.tangent.w;

                float3x3 TBN = float3x3(tangent, binormal, v.normal);

                o.LightDir = normalize(mul(TBN, ObjSpaceLightDir(v.vertex))).xyz;

                o.ViewDir = normalize(mul(TBN, ObjSpaceViewDir(v.vertex))).xyz;

                return o;

            }

  

            fixed4 frag (v2f i) : SV_Target

            {

                // sample the texture

                fixed3 tangentLightDir = normalize(i.LightDir);

                fixed3 tangentViewDir = normalize(i.ViewDir);

                fixed4 packNormal = tex2D(_BumpMap, i.uv);

                fixed3 tangentNormal = UnpackNormal(packNormal);

                tangentNormal.xy *= _BumpScale;

                tangentNormal.z = sqrt(1.0 - saturate(dot(tangentNormal.xy, tangentNormal.xy)));

                fixed3 albedo = tex2D(_MainTex, i.uv).rgb;

                fixed3 ambient = UNITY_LIGHTMODEL_AMBIENT.xyz * albedo;

                fixed3 diffuse = _LightColor0.rgb * albedo * max(0.0, dot(tangentNormal, tangentLightDir));

                float3 halfDir = normalize(tangentLightDir + tangentViewDir);

                fixed3 specular = _LightColor0.rgb * _Specular.rgb * pow(max(0.0, dot(tangentNormal, halfDir)), _Gloss);

                return fixed4(ambient + diffuse + specular, 1.0);

            }

##### 第二种：世界空间下的法线：
struct appdata

            {

                float4 vertex : POSITION;

                float3 normal : NORMAL;

                float4 tangent : TANGENT;

                float2 uv : TEXCOORD0;

            };

  

            struct v2f

            {

                float4 pos : SV_POSITION;

                float4 uv : TEXCOORD0;

                float3 worldNormal : TEXCOORD1;

                float3 worldTangent : TEXCOORD2;

                float3 worldBinormal : TEXCOORD3;

                float3 worldPos : TEXCOORD4;

            };

  

            sampler2D _MainTex;

            sampler2D _BumpMap;

            float4 _MainTex_ST;

            float4 _BumpMap_ST;

            float4 _Specular;

            float _Gloss;

            float _BumpScale;

  

            v2f vert (appdata v)

            {

              v2f o;

              o.pos = UnityObjectToClipPos(v.vertex);

              o.uv.xy = v.uv.xy * _MainTex_ST.xy + _MainTex_ST.zw;

              o.uv.zw = v.uv.xy * _BumpMap_ST.xy + _BumpMap_ST.zw;

              o.worldNormal = UnityObjectToWorldNormal(v.normal);

              o.worldTangent = UnityObjectToWorldDir(v.tangent.xyz);

              o.worldBinormal = cross(o.worldNormal, o.worldTangent) * v.tangent.w;

              o.worldPos = mul(unity_ObjectToWorld, v.vertex).xyz;

              return o;

            }

  

            fixed4 frag (v2f i) : SV_Target

            {

              float3 worldPos = i.worldPos;

              float3 worldNormal = normalize(i.worldNormal);

              fixed3 lightDir = normalize(UnityWorldSpaceLightDir(worldPos));

              fixed3 viewDir = normalize(UnityWorldSpaceViewDir(worldPos));

              fixed3 bump = UnpackNormal(tex2D(_BumpMap, i.uv.zw));

              bump.xy *= _BumpScale;

              bump.z = sqrt(1.0 - saturate(dot(bump.xy, bump.xy)));

              bump = normalize(bump.x * normalize(i.worldTangent) + bump.y * normalize(i.worldBinormal) + bump.z * normalize(i.worldNormal));

              fixed3 albedo = tex2D(_MainTex, i.uv.xy).rgb * _LightColor0.rgb;

              fixed3 ambient = UNITY_LIGHTMODEL_AMBIENT.xyz * albedo;

              fixed3 diffuse = albedo * max(0,dot(bump, lightDir));

              fixed3 halfDir = normalize(lightDir + viewDir);

              fixed3 specular = _Specular.rgb * pow(max(0,dot(bump, halfDir)), _Gloss);

              return fixed4(ambient + diffuse + specular, 1.0);

            }
